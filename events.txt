CREATE TABLE events (
    evCode INT AUTO_INCREMENT PRIMARY KEY,
    evName VARCHAR(150) NOT NULL,
    evDate DATE NOT NULL,
    evVenue VARCHAR(150) NOT NULL,
    evRFee DECIMAL(10,2) NOT NULL
);

CREATE TABLE participants (
    partID INT AUTO_INCREMENT PRIMARY KEY,
    evCode INT NOT NULL,
    fName VARCHAR(100) NOT NULL,
    lName VARCHAR(100) NOT NULL,
    partRate DECIMAL(5,2) DEFAULT 0,
    FOREIGN KEY (evCode) REFERENCES events(evCode)
);

CREATE TABLE registration (
    regCode INT AUTO_INCREMENT PRIMARY KEY,
    partID INT NOT NULL,
    regDate DATE NOT NULL,
    regFeePaid DECIMAL(10,2) NOT NULL,
    regPayMode VARCHAR(20) NOT NULL,
    FOREIGN KEY (partID) REFERENCES participants(partID)
);

# EVENTS MANAGEMENT
def events_management():

    while True:
        print("\n===== EVENTS MANAGEMENT =====")
        print("1. Add Event")
        print("2. View Events")
        print("3. Search Event")
        print("4. Update Event")
        print("5. Delete Event")
        print("6. Back to Main Menu")

        choice = input("\nEnter your choice: ")

        # ADD EVENT
        if choice == "1":

            evName = input("Enter Event Name: ")
            evDate = input("Enter Event Date (YYYY-MM-DD): ")
            evVenue = input("Enter Event Venue: ")
            evRFee = float(input("Enter Registration Fee: "))

            sql = """
                INSERT INTO events
                (evName, evDate, evVenue, evRFee)
                VALUES (%s, %s, %s, %s)
            """

            values = (evName, evDate, evVenue, evRFee)

            cursor.execute(sql, values)
            conn.commit()

            print("\nEvent added successfully!")

        # VIEW EVENTS
        elif choice == "2":

            sql = "SELECT * FROM events"
            cursor.execute(sql)

            records = cursor.fetchall()

            print("\n========== EVENT RECORDS ==========")

            if records:
                for row in records:
                    print("--------------------------------")
                    print("Event Code:", row[0])
                    print("Event Name:", row[1])
                    print("Event Date:", row[2])
                    print("Event Venue:", row[3])
                    print("Registration Fee:", row[4])
            else:
                print("No events found.")

        # SEARCH EVENT
        elif choice == "3":

            evCode = input("\nEnter Event Code to search: ")

            sql = "SELECT * FROM events WHERE evCode = %s"
            values = (evCode,)

            cursor.execute(sql, values)
            record = cursor.fetchone()

            if record:
                print("\n========== EVENT FOUND ==========")
                print("Event Code:", record[0])
                print("Event Name:", record[1])
                print("Event Date:", record[2])
                print("Event Venue:", record[3])
                print("Registration Fee:", record[4])
            else:
                print("\nEvent not found!")

        # UPDATE EVENT
        elif choice == "4":

            evCode = input("Enter Event Code to update: ")

            sql = "SELECT * FROM events WHERE evCode = %s"
            cursor.execute(sql, (evCode,))

            record = cursor.fetchone()

            if record:

                print("\n========== CURRENT EVENT DETAILS ==========")
                print("Event Name:", record[1])
                print("Event Date:", record[2])
                print("Event Venue:", record[3])
                print("Registration Fee:", record[4])

                evName = input("Enter New Event Name: ")
                evDate = input("Enter New Event Date (YYYY-MM-DD): ")
                evVenue = input("Enter New Event Venue: ")
                evRFee = float(input("Enter New Registration Fee: "))

                sql = """
                    UPDATE events
                    SET evName = %s,
                        evDate = %s,
                        evVenue = %s,
                        evRFee = %s

                cursor.execute(sql, values)

            else:
                print("\nEvent not found!")

        # DELETE EVENT

            evCode = input("Enter Event Code to delete: ")

            sql = "SELECT * FROM events WHERE evCode = %s"
            cursor.execute(sql, (evCode,))

            record = cursor.fetchone()

            if record:

                print("\n========== EVENT DETAILS ==========")
                print("Event Code:", record[0])
                print("Event Name:", record[1])

                confirm = input("\nAre you sure you want to delete this event? (Y/N): ")

                    cursor.execute(sql, (evCode,))
                    conn.commit()

                else:
                    print("\nDeletion cancelled.")

            else:
                print("\nEvent not found!")

                p.partID,

                e.evName,

            INNER JOIN events e

            print("Final Registration Fee:", regFeePaid)

                "Enter Payment Mode (Cash/GCash/Bank): "
            )

            sql = """
                INSERT INTO registration
                (partID, regDate, regFeePaid, regPayMode)

                partID,
                regDate,

def registration_monitoring():

    print("3. View Registration Summary")
    print("4. Back to Main Menu")

        sql = """
            SELECT
                r.regCode,
                p.partID,
                p.partFName,
                p.partLName,
                e.evName,
                e.evDate,
                e.evVenue,
                r.regDate,
                r.regFeePaid,
                r.regPayMode
            FROM registration r
            INNER JOIN participants p
            ON r.partID = p.partID
            INNER JOIN events e
            ON p.evCode = e.evCode
        """

        cursor.execute(sql)
        records = cursor.fetchall()

        print("\n========== ALL REGISTRATIONS ==========")

        if records:

            for row in records:

                print("--------------------------------")

                print("Participant ID:", row[1])
                print("Participant Name:", row[2], row[3])
                print("Event Name:", row[4])
                print("Event Date:", row[5])
                print("Event Venue:", row[6])
                print("Registration Date:", row[7])
                print("Registration Fee Paid:", row[8])
                print("Payment Mode:", row[9])

        else:
            print("No registrations found.")


    # FILTER BY EVENT NAME
    elif choice == "2":

        evName = input("\nEnter Event Name to filter: ")

        sql = """
            SELECT
                r.regCode,
                p.partID,
                p.partFName,
                p.partLName,
                e.evName,
                r.regDate,
                r.regFeePaid,
                r.regPayMode
            FROM registration r
            INNER JOIN participants p
            ON r.partID = p.partID
            INNER JOIN events e
            ON p.evCode = e.evCode
            WHERE e.evName LIKE %s
        """

        values = ("%" + evName + "%",)

        cursor.execute(sql, values)
        records = cursor.fetchall()

        print("\n========== FILTERED REGISTRATIONS ==========")

        if records:

            for row in records:

                print("--------------------------------")
                print("Registration Code:", row[0])
                print("Participant ID:", row[1])
                print("Participant Name:", row[2], row[3])
                print("Event Name:", row[4])
                print("Registration Date:", row[5])
                print("Fee Paid:", row[6])
                print("Payment Mode:", row[7])

        else:
            print("No registrations found for this event.")


    # VIEW REGISTRATION SUMMARY
    elif choice == "3":

        sql = """
            SELECT
                e.evCode,
                e.evName,
                COUNT(r.regCode),
                COALESCE(SUM(r.regFeePaid), 0)
            FROM events e
            LEFT JOIN participants p
            ON e.evCode = p.evCode
            LEFT JOIN registration r
            ON p.partID = r.partID
