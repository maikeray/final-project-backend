def borrowing_management():

    while True:

        print("\n===== BOOK BORROWING SYSTEM =====")
        print("1. Check Book Availability")
        print("2. Borrow Book")
        print("3. View Borrowed Books")
        print("4. Return Book")
        print("5. View Returned Books")
        print("6. Back to Main Menu")

        choice = input("\nEnter your choice: ")

        # CHECK BOOK AVAILABILITY
        if choice == "1":

            print("\n===== BOOK AVAILABILITY =====")

            bookID = int(input("Enter Book ID: "))

            cursor.execute(
                "SELECT * FROM book WHERE bookID = %s",
                (bookID,)
            )

            book = cursor.fetchone()

            if book is None:
                print("Book not found!")

            else:

                cursor.execute(
                    """
                    SELECT COUNT(*) FROM borrowed_book
                    WHERE bookID = %s AND status = 'Borrowed'
                    """,
                    (bookID,)
                )

                borrowed = cursor.fetchone()[0]

                available = book[2] - borrowed

                print("\nBook Title:", book[1])
                print("Total Copies:", book[2])
                print("Borrowed Copies:", borrowed)
                print("Available Copies:", available)

        # BORROW BOOK
        elif choice == "2":

            print("\n===== BORROW BOOK =====")

            studID = int(input("Enter Student ID: "))
            bookID = int(input("Enter Book ID: "))

            cursor.execute(
                "SELECT * FROM student WHERE studID = %s",
                (studID,)
            )

            student = cursor.fetchone()

            cursor.execute(
                "SELECT * FROM book WHERE bookID = %s",
                (bookID,)
            )

            book = cursor.fetchone()

            if student is None:
                print("Student not found!")

            elif book is None:
                print("Book not found!")

            else:

                cursor.execute(
                    """
                    SELECT COUNT(*) FROM borrowed_book
                    WHERE bookID = %s AND status = 'Borrowed'
                    """,
                    (bookID,)
                )

                borrowed = cursor.fetchone()[0]

                available = book[2] - borrowed

                if available <= 0:
                    print("No available copies of this book!")

                else:

                    sql = """
                    INSERT INTO borrowed_book
                    (studID, bookID, status)
                    VALUES (%s, %s, %s)
                    """

                    cursor.execute(
                        sql,
                        (studID, bookID, "Borrowed")
                    )

                    conn.commit()

                    print("\nBook borrowed successfully!")
                    print("Borrow ID:", cursor.lastrowid)

        # VIEW BORROWED BOOKS
        elif choice == "3":

            print("\n===== BORROWED BOOK RECORDS =====")

            sql = """
            SELECT borrowed_book.borrowID,
                   student.studID,
                   student.studFName,
                   student.studLName,
                   book.bookID,
                   book.bookTitle,
                   borrowed_book.status
            FROM borrowed_book
            INNER JOIN student
            ON borrowed_book.studID = student.studID
            INNER JOIN book
            ON borrowed_book.bookID = book.bookID
            WHERE borrowed_book.status = 'Borrowed'
            """

            cursor.execute(sql)

            results = cursor.fetchall()

            if len(results) == 0:
                print("No borrowed books found.")

            else:
                for record in results:
                    print("-------------------------")
                    print("Borrow ID:", record[0])
                    print("Student ID:", record[1])
                    print("Student Name:", record[2], record[3])
                    print("Book ID:", record[4])
                    print("Book Title:", record[5])
                    print("Status:", record[6])

        # RETURN BOOK
        elif choice == "4":

            print("\n===== RETURN BOOK =====")

            borrowID = int(input("Enter Borrow ID: "))

            cursor.execute(
                """
                SELECT * FROM borrowed_book
                WHERE borrowID = %s
                """,
                (borrowID,)
            )

            record = cursor.fetchone()

            if record is None:
                print("Borrow record not found!")

            elif record[3] == "Returned":
                print("This book has already been returned!")

            else:

                cursor.execute(
                    """
                    UPDATE borrowed_book
                    SET status = 'Returned'
                    WHERE borrowID = %s
                    """,
                    (borrowID,)
                )

                conn.commit()

                print("\nBook returned successfully!")

        # VIEW RETURNED BOOKS
        elif choice == "5":

            print("\n===== RETURNED BOOK RECORDS =====")

            sql = """
            SELECT borrowed_book.borrowID,
                   student.studFName,
                   student.studLName,
                   book.bookTitle,
                   borrowed_book.status
            FROM borrowed_book
            INNER JOIN student
            ON borrowed_book.studID = student.studID
            INNER JOIN book
            ON borrowed_book.bookID = book.bookID
            WHERE borrowed_book.status = 'Returned'
            """

            cursor.execute(sql)

            results = cursor.fetchall()

            if len(results) == 0:
                print("No returned books found.")

            else:
                for record in results:
                    print("-------------------------")
                    print("Borrow ID:", record[0])
                    print("Student Name:", record[1], record[2])
                    print("Book Title:", record[3])
                    print("Status:", record[4])

        elif choice == "6":
            break

        else:
            print("Invalid choice!")
