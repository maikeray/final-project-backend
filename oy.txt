@app.route("/registration/add", methods=["POST"])
def add_registration():

    partID = request.form["partID"]
    regDate = request.form["regDate"]
    regPayMode = request.form["regPayMode"]

    conn = get_db_connection()
    cursor = conn.cursor()

    try:
        # GET PARTICIPANT'S EVENT FEE AND DISCOUNT
        cursor.execute("""
            SELECT e.evRFee, p.partDRate
            FROM participants p
            JOIN events e ON p.evCode = e.evCode
            WHERE p.partID = %s
        """, (partID,))

        participant = cursor.fetchone()

        if participant is None:
            flash("Participant not found!")

        else:
            event_fee = float(participant[0])
            discount_rate = float(participant[1])

            discount = event_fee * discount_rate / 100
            final_fee = event_fee - discount

            # SAVE REGISTRATION
            sql = """
                INSERT INTO registration
                (partID, regDate, regFeePaid, regPayMode)
                VALUES (%s, %s, %s, %s)
            """

            values = (
                partID,
                regDate,
                final_fee,
                regPayMode
            )

            cursor.execute(sql, values)
            conn.commit()

            flash(
                f"Registration successful! "
                f"Final fee: ₱{final_fee:.2f}"
            )

    except Exception as e:
        conn.rollback()
        flash(f"Error registering participant: {e}")

    finally:
        cursor.close()
        conn.close()

    return redirect(url_for("registration"))

@app.route("/monitoring")
def monitoring():

    selected_evCode = request.args.get("evCode", type=int)

    conn = get_db_connection()
    cursor = conn.cursor()

    try:
        # GET ALL EVENTS FOR THE FILTER
        cursor.execute("""
            SELECT evCode, evName
            FROM events
            ORDER BY evName
        """)

        events_list = cursor.fetchall()

        # DEFAULT AGGREGATE VALUES
        registrations = []
        total_count = 0
        total_fees = 0
        total_discounts = 0

        # GET REGISTRATION RECORDS FOR SELECTED EVENT
        if selected_evCode is not None:

            cursor.execute("""
                SELECT
                    e.evName,
                    CONCAT(p.partFName, ' ', p.partLName),
                    r.regDate,
                    r.regFeePaid
                FROM registration r
                JOIN participants p ON r.partID = p.partID
                JOIN events e ON p.evCode = e.evCode
                WHERE e.evCode = %s
                ORDER BY r.regDate DESC, r.regCode DESC
            """, (selected_evCode,))

            registrations = cursor.fetchall()

            # GET AGGREGATE VALUES
            cursor.execute("""
                SELECT
                    COUNT(r.regCode),
                    COALESCE(SUM(r.regFeePaid), 0),
                    (
                        COUNT(r.regCode) * e.evRFee
                        - COALESCE(SUM(r.regFeePaid), 0)
                    )
                FROM events e
                LEFT JOIN participants p ON e.evCode = p.evCode
                LEFT JOIN registration r ON p.partID = r.partID
                WHERE e.evCode = %s
                GROUP BY e.evCode, e.evRFee
            """, (selected_evCode,))

            summary = cursor.fetchone()

            if summary:
                total_count = summary[0]
                total_fees = summary[1]
                total_discounts = summary[2]

    finally:
        cursor.close()
        conn.close()

    return render_template(
        "monitoring.html",
        events=events_list,
        registrations=registrations,
        selected_evCode=selected_evCode,
        total_count=total_count,
        total_fees=total_fees,
        total_discounts=total_discounts
    )
